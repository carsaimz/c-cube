    64→function easeOutCubic(t: number): number {
    65→  return 1 - Math.pow(1 - t, 3);
    66→}
    67→
    68→function easeOutQuad(t: number): number {
    69→  return 1 - (1 - t) * (1 - t);
    70→}
    71→
    72→function easeInOutCubic(t: number): number {
    73→  return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;

    76→function defaultDuration(m: Move, queueLen: number): number {
    77→  switch (m.source) {
    78→    case "user":
    79→      return 200;
    80→    case "undo":
    81→    case "redo":
    82→      return 170;
    83→    case "scramble":
    84→      return 95;
    85→    case "autosolve":
    86→      return queueLen > 24 ? 65 : 85;
    87→  }

   200→  private makeStickerTexture(faceIndex: number): THREE.Texture {
   201→    const key = `${this.theme.id}-${faceIndex}`;
   202→    const cached = this.textureCache.get(key);
   203→    if (cached) return cached;
   204→    const canvas = document.createElement("canvas");
   205→    canvas.width = 256;
   206→    canvas.height = 256;
   207→    const g = canvas.getContext("2d");
   208→    const color = this.theme.faces[faceIndex];
   209→    if (g) {
   210→      const r = 62;
   211→      const m = 10;
   212→      g.clearRect(0, 0, 256, 256);
   213→      // gradiente radial suave (brilho no topo-esquerda)
   214→      const grad = g.createRadialGradient(96, 78, 20, 128, 128, 190);
   215→      grad.addColorStop(0, lighten(color, 0.22));
   216→      grad.addColorStop(0.55, color);
   217→      grad.addColorStop(1, lighten(color, -0.14));
   218→      g.fillStyle = grad;
   219→      roundRect(g, m, m, 256 - 2 * m, 256 - 2 * m, r);
   220→      g.fill();
   221→      // contorno interior discreto
   222→      g.strokeStyle = "rgba(0,0,0,0.18)";
   223→      g.lineWidth = 5;
   224→      roundRect(g, m + 3, m + 3, 256 - 2 * m - 6, 256 - 2 * m - 6, r - 4);
   225→      g.stroke();
   226→      // reflexo superior
   227→      g.fillStyle = "rgba(255,255,255,0.13)";
   228→      roundRect(g, m + 14, m + 12, 256 - 2 * m - 28, 70, 44);
   229→      g.fill();
   230→    }
   231→    const tex = new THREE.CanvasTexture(canvas);
   232→    tex.colorSpace = THREE.SRGBColorSpace;
   233→    tex.anisotropy = 4;
   234→    this.textureCache.set(key, tex);
   235→    return tex;
   236→  }

   243→  private disposeCubeObjects(): void {
   244→    for (const c of this.cubies) {
   245→      c.group.traverse((o) => {
   246→        const mesh = o as THREE.Mesh;
   247→        if (mesh.isMesh) {
   248→          const mat = mesh.material as THREE.Material;
   249→          if (mat && !Array.isArray(mat) && mat !== (this.hitboxMaterial as THREE.Material)) mat.dispose();
   250→        }
   251→      });
   252→      this.cubeGroup.remove(c.group);
   253→    }
   254→    this.cubies = [];
   255→    this.stickerMeshes = [];
   256→    this.hitMeshes = [];
   257→  }

   259→  private buildCube(): void {
   260→    this.disposeCubeObjects();
   261→    const n = this.size;
   262→    const c = (n - 1) / 2;
   263→    const plasticMat = new THREE.MeshStandardMaterial({
   264→      color: new THREE.Color(this.theme.plastic),
   265→      roughness: 0.34,
   266→      metalness: 0.22,
   267→    });
   268→    for (let x = 0; x < n; x++) {
   269→      for (let y = 0; y < n; y++) {
   270→        for (let z = 0; z < n; z++) {
   271→          const onSurface = x === 0 || x === n - 1 || y === 0 || y === n - 1 || z === 0 || z === n - 1;
   272→          if (!onSurface) continue;
   273→          const group = new THREE.Group();
   274→          const body = new THREE.Mesh(this.bodyGeometry, plasticMat);
   275→          group.add(body);
   276→          const hitbox = new THREE.Mesh(this.hitboxGeometry, this.hitboxMaterial);
   277→          group.add(hitbox);
   278→          const pos: Vec3 = [x, y, z];
   279→          const stickerDirs: Vec3[] = [];
   280→          const dirs: Vec3[] = [
   281→            [1, 0, 0],
   282→            [-1, 0, 0],
   283→            [0, 1, 0],
   284→            [0, -1, 0],
   285→            [0, 0, 1],
   286→            [0, 0, -1],
   287→          ];
   288→          dirs.forEach((d, faceIndex) => {
   289→            const onBoundary =
   290→              (d[0] === 1 && x === n - 1) ||
   291→              (d[0] === -1 && x === 0) ||
   292→              (d[1] === 1 && y === n - 1) ||
   293→              (d[1] === -1 && y === 0) ||
   294→              (d[2] === 1 && z === n - 1) ||
   295→              (d[2] === -1 && z === 0);
   296→            if (!onBoundary) return;
   297→            stickerDirs.push(d);
   298→            const faceIndex2 =
   299→              d[0] === 1 ? 0 : d[0] === -1 ? 1 : d[1] === 1 ? 2 : d[1] === -1 ? 3 : d[2] === 1 ? 4 : 5;
   300→            const mat = new THREE.MeshStandardMaterial({
   301→              map: this.makeStickerTexture(faceIndex2),
   302→              roughness: 0.38,
   303→              metalness: 0.05,
   304→            });
   305→            if (this.theme.glow > 0) {
   306→              mat.emissive = new THREE.Color(0xffffff);
   307→              mat.emissiveMap = mat.map ?? null;
   308→              mat.emissiveIntensity = this.theme.glow;
   309→            }
   310→            const sticker = new THREE.Mesh(this.stickerGeometry, mat);
   311→            const v = new THREE.Vector3(d[0], d[1], d[2]);
   312→            sticker.position.set(d[0] * 0.476, d[1] * 0.476, d[2] * 0.476);
   313→            sticker.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), v);
   314→            sticker.userData.cubie = null; // preenchido abaixo
   315→            group.add(sticker);
   316→            this.stickerMeshes.push(sticker);
   317→          });
   318→          const cubie: Cubie = {
   319→            pos,
   320→            stickerDirs,
   321→            M: [
   322→              [1, 0, 0],
   323→              [0, 1, 0],
   324→              [0, 0, 1],
   325→            ],
   326→            group,
   327→          };
   328→          group.position.set((x - c) * this.spacing, (y - c) * this.spacing, (z - c) * this.spacing);
   329→          group.userData.cubie = cubie;
   330→          for (const child of group.children) {
   331→            if (child instanceof THREE.Mesh && child !== body) child.userData.cubie = cubie;
   332→          }
   333→          this.hitMeshes.push(hitbox);
   334→          this.cubies.push(cubie);
   335→          this.cubeGroup.add(group);
   336→        }
   337→      }
   338→    }
   339→    this.buildShadow();
   340→  }

   373→  private buildShadow(): void {
   374→    if (this.shadowMesh) {
   375→      this.scene.remove(this.shadowMesh);
   376→      this.shadowMesh = null;
   377→    }
   378→    if (!this.shadowTexture) {
   379→      const canvas = document.createElement("canvas");
   380→      canvas.width = 256;
   381→      canvas.height = 256;
   382→      const g = canvas.getContext("2d");
   383→      if (g) {
   384→        const grad = g.createRadialGradient(128, 128, 10, 128, 128, 124);
   385→        grad.addColorStop(0, "rgba(0,0,0,0.55)");
   386→        grad.addColorStop(1, "rgba(0,0,0,0)");
   387→        g.fillStyle = grad;
   388→        g.fillRect(0, 0, 256, 256);
   389→      }
   390→      const tex = new THREE.CanvasTexture(canvas);
   391→      this.shadowTexture = tex;
   392→    }
   393→    const geo = new THREE.PlaneGeometry(this.size * 1.9, this.size * 1.9);
   394→    const mat = new THREE.MeshBasicMaterial({
   395→      map: this.shadowTexture,
   396→      transparent: true,
   397→      opacity: 0.5,
   398→      depthWrite: false,
   399→    });
   400→    const mesh = new THREE.Mesh(geo, mat);
   401→    mesh.rotation.x = -HALF_PI;
   402→    mesh.position.y = -(this.size - 1) / 2 * this.spacing - 0.9;
   403→    this.shadowMesh = mesh;
   404→    this.scene.add(mesh);
   405→  }

   407→  private applyThemeToMaterials(): void {
   408→    for (const c of this.cubies) {
   409→      c.group.traverse((o) => {
   410→        const mesh = o as THREE.Mesh;
   411→        if (!mesh.isMesh) return;
   412→        const mat = mesh.material as THREE.MeshStandardMaterial;
   413→        if (!mat || Array.isArray(mat)) return;
   414→        if (mat.map) {
   415→          if (this.blind) {
   416→            // modo memória: adesivos neutros, sem brilho
   417→            mat.map = this.makeBlindTexture();
   418→            mat.emissive = new THREE.Color(0x000000);
   419→            mat.emissiveMap = null;
   420→            mat.emissiveIntensity = 0;
   421→            mat.needsUpdate = true;
   422→            return;
   423→          }
   424→          // adesivo: descobrir face pela orientação local
   425→          const dir = new THREE.Vector3(0, 0, 1).applyQuaternion(mesh.quaternion);
   426→          const faceIndex =
   427→            dir.x > 0.5 ? 0 : dir.x < -0.5 ? 1 : dir.y > 0.5 ? 2 : dir.y < -0.5 ? 3 : dir.z > 0.5 ? 4 : 5;
   428→          mat.map = this.makeStickerTexture(faceIndex);
   429→          if (this.theme.glow > 0) {
   430→            mat.emissive = new THREE.Color(0xffffff);
   431→            mat.emissiveMap = mat.map;
   432→            mat.emissiveIntensity = this.theme.glow;
   433→          } else {
   434→            mat.emissive = new THREE.Color(0x000000);
   435→            mat.emissiveMap = null;
   436→            mat.emissiveIntensity = 0;
   437→          }
   438→          mat.needsUpdate = true;
   439→        } else {
   440→          mat.color = new THREE.Color(this.theme.plastic);
   441→        }
   442→      });
   443→    }
   444→  }

   448→  private fitCamera(): void {
   449→    const r = this.idealCameraDistance();
   450→    this.camera.position.set(r * 0.66, r * 0.5, r * 0.82);
   451→    this.camera.lookAt(0, 0, 0);
   452→    this.controls.target.set(0, 0, 0);
   453→    this.controls.minDistance = this.size * 1.7 + 0.8;
   454→    this.controls.maxDistance = r * 1.9;
   455→    this.controls.update();
   456→    this.updateFrameShiftTarget();
   457→  }

   459→  /** Distância vertical (px) entre o centro do ecrã e o centro da zona livre entre HUD e dock. */
   460→  private updateFrameShiftTarget(): void {
   461→    if (!this.framed) {
   462→      this.frameShiftTarget = 0;
   463→      return;
   464→    }
   465→    const h = Math.max(1, this.container.clientHeight);
   466→    // HUD superior ~8% da altura, dock inferior ~13% (com mínimos em px)
   467→    const top = Math.max(56, h * 0.085);
   468→    const bottom = Math.max(96, h * 0.14);
   469→    this.frameShiftTarget = Math.max(0, (bottom - top) / 2);
   470→  }

   472→  /**
   473→   * Repõe a câmara: centro (0,0,0), zoom padrão e orientação de origem, com
   474→   * animação suave. Chamado pelo botão "Centrar" e por duplo clique no fundo.
   475→   */
   476→  resetView(): void {
   477→    const r = this.idealCameraDistance();
   478→    this.defaultCamPos.set(r * 0.66, r * 0.5, r * 0.82);
   479→    this.controls.minDistance = this.size * 1.7 + 0.8;
   480→    this.controls.maxDistance = r * 1.9;
   481→    this.camTween = {
   482→      fromPos: this.camera.position.clone(),
   483→      toPos: this.defaultCamPos.clone(),
   484→      fromTarget: this.controls.target.clone(),
   485→      toTarget: new THREE.Vector3(0, 0, 0), // o offset visual (setViewOffset) trata do enquadramento
   486→      start: performance.now(),
   487→      duration: 650,
   488→    };
   489→    // repe a deriva do grupo do cubo (flutuação/rotação do menu)
   490→    this.cubeGroup.position.y = 0;
   491→    this.cubeGroup.rotation.x = 0;
   492→    this.cubeGroup.rotation.z = 0;
   493→  }

   495→  /** Distância ideal da câmara para o tamanho do cubo + aspeto atual do contentor. */
   496→  private idealCameraDistance(): number {
   497→    const s = this.size;
   498→    const base = s * 2.45 + 1.75;
   499→    const aspect = this.container.clientWidth / Math.max(1, this.container.clientHeight);
   500→    const halfW = (s * 1.04) / 2 + 0.25;
   501→    const hTan = Math.tan((this.camera.fov * Math.PI) / 360) * Math.max(0.3, aspect);
   502→    const need = (halfW / hTan) * 1.06;
   503→    return Math.min(Math.max(base, need), s * 7 + 4);
   504→  }

   506→  private onDblClick = (e: MouseEvent): void => {
   507→    // duplo clique em zona vazia (fora do cubo) → recentrar
   508→    this.updatePointerNDC(e as unknown as PointerEvent);
   509→    this.raycaster.setFromCamera(this.pointer, this.camera);
   510→    if (this.raycaster.intersectObjects(this.hitMeshes, false).length === 0) {
   511→      this.resetView();
   512→    }
   513→  };

   532→  private updatePointerNDC(e: PointerEvent): void {
   533→    const rect = this.renderer.domElement.getBoundingClientRect();
   534→    this.pointer.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
   535→    this.pointer.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
   536→  }

   538→  private onPointerDown = (e: PointerEvent): void => {
   539→    if (e.button !== 0 && e.pointerType === "mouse") return;
   540→    if (this.anim || this.queue.length > 0) return;
   541→    this.updatePointerNDC(e);
   542→    this.raycaster.setFromCamera(this.pointer, this.camera);
   543→    const hits = this.raycaster.intersectObjects(this.hitMeshes, false);
   544→    if (hits.length === 0) return;
   545→    const hit = hits[0];
   546→    const cubie = hit.object.userData.cubie as Cubie | undefined;
   547→    if (!cubie) return;
   548→    // normal da face da hitbox -> espaço do mundo
   549→    const worldNormal = hit.face
   550→      ? hit.face.normal.clone().transformDirection(hit.object.matrixWorld)
   551→      : hit.object.getWorldDirection(new THREE.Vector3());
   552→    const snapped = snapAxis(worldNormal);
   553→    this.drag = {
   554→      x0: e.clientX,
   555→      y0: e.clientY,
   556→      cubie,
   557→      normal: snapped,
   558→      point: hit.point.clone(),
   559→      decided: false,
   560→    };
   561→    this.controls.enabled = false;
   562→  };
   563→
   564→  private onPointerMove = (e: PointerEvent): void => {
   565→    if (!this.drag || this.drag.decided) return;
   566→    const dx = e.clientX - this.drag.x0;
   567→    const dy = e.clientY - this.drag.y0;
   568→    if (Math.hypot(dx, dy) < 12) return;
   569→
   570→    // vetor de arrasto no mundo (plano do adesivo)
   571→    this.updatePointerNDC(e);
   572→    this.raycaster.setFromCamera(this.pointer, this.camera);
   573→    const n = new THREE.Vector3(...this.drag.normal);
   574→    const plane = new THREE.Plane().setFromNormalAndCoplanarPoint(n, this.drag.point);
   575→    const pNow = new THREE.Vector3();
   576→    let delta: THREE.Vector3 | null = this.raycaster.ray.intersectPlane(plane, pNow)
   577→      ? pNow.clone().sub(this.drag.point)
   578→      : null;
   579→    if (!delta || delta.lengthSq() < 1e-8) {
   580→      // plano quase de perfil: aproximar pelo ecrã
   581→      const dist = this.camera.position.distanceTo(this.drag.point);
   582→      const k = dist * 0.0016;
   583→      const right = new THREE.Vector3().setFromMatrixColumn(this.camera.matrixWorld, 0);
   584→      const up = new THREE.Vector3().setFromMatrixColumn(this.camera.matrixWorld, 1);
   585→      delta = right.multiplyScalar(dx * k).add(up.multiplyScalar(-dy * k));
   586→    }
   587→    delta.normalize();
   588→
   589→    // eixos candidatos: os dois perpendiculares à normal do adesivo
   590→    const normalAxis = axisOf(this.drag.normal);
   591→    const r = this.drag.point.clone().sub(this.cubeGroup.position);
   592→    let best: { axis: Axis; dir: 1 | -1; score: number } | null = null;
   593→    for (const axis of ["x", "y", "z"] as Axis[]) {
   594→      if (axis === normalAxis) continue;
   595→      for (const dir of [1, -1] as const) {
   596→        const omega = AXIS_VEC[axis].clone().multiplyScalar(dir);
   597→        const v = new THREE.Vector3().crossVectors(omega, r).normalize();
   598→        const score = v.dot(delta);
   599→        if (!best || score > best.score) best = { axis, dir, score };
   600→      }
   601→    }
   602→    this.drag.decided = true;
   603→    if (best && best.score > 0.12) {
   604→      const layer = this.drag.cubie.pos[AXIS_INDEX[best.axis]];
   605→      this.cb.onUserGestureMove({
   606→        axis: best.axis,
   607→        layer,
   608→        dir: best.dir,
   609→        source: "user",
   610→      });
   611→    }
   612→  };
   613→
   614→  private onPointerUp = (): void => {
   615→    if (this.drag) {
   616→      this.drag = null;
   617→      this.controls.enabled = true;
   618→    }
   619→  };

   643→  private startNext(): void {
   644→    const move = this.queue.shift();
   645→    if (!move) return;
   646→    const ai = AXIS_INDEX[move.axis];
   647→    const members = this.cubies.filter((c) => c.pos[ai] === move.layer);
   648→    this.pivot.rotation.set(0, 0, 0);
   649→    this.pivot.updateMatrixWorld(true);
   650→    for (const c of members) this.pivot.attach(c.group);
   651→    this.anim = {
   652→      move,
   653→      cubies: members,
   654→      start: performance.now(),
   655→      duration: move.duration ?? defaultDuration(move, this.queue.length),
   656→    };
   657→    this.cb.onMoveStart(move);
   658→  }
   659→
   660→  private commitCurrent(): void {
   661→    const { move, cubies } = this.anim!;
   662→    this.pivot.rotation.set(0, 0, 0);
   663→    (this.pivot.rotation as unknown as Record<string, number>)[move.axis] = move.dir * HALF_PI;
   664→    this.pivot.updateMatrixWorld(true);
   665→    const R = rotationMatrix(move.axis, move.dir);
   666→    const c = (this.size - 1) / 2;
   667→    const m4 = new THREE.Matrix4();
   668→    for (const cubie of cubies) {
   669→      this.cubeGroup.attach(cubie.group);
   670→      cubie.pos = rotateLogicalPos(cubie.pos, move.axis, move.dir, this.size);
   671→      cubie.M = matMul(R, cubie.M);
   672→      // encaixe exato (elimina deriva numérica)
   673→      cubie.group.position.set(
   674→        (cubie.pos[0] - c) * this.spacing,
   675→        (cubie.pos[1] - c) * this.spacing,
   676→        (cubie.pos[2] - c) * this.spacing
   677→      );
   678→      const M = cubie.M;
   679→      m4.set(M[0][0], M[0][1], M[0][2], 0, M[1][0], M[1][1], M[1][2], 0, M[2][0], M[2][1], M[2][2], 0, 0, 0, 0, 1);
   680→      cubie.group.quaternion.setFromRotationMatrix(m4);
   681→      cubie.group.scale.set(1, 1, 1);
   682→    }
   683→    this.pivot.rotation.set(0, 0, 0);
   684→    const committed = this.anim!.move;
   685→    this.anim = null;
   686→    this.cb.onMoveCommitted(committed);
   687→  }
   688→
   689→  private tick = (): void => {
   690→    if (this.disposed) return;
   691→    const now = performance.now();
   692→    const t = now - this.clockStart;
   693→
   694→    // tween da câmara (botão centrar / duplo clique)
   695→    if (this.camTween) {
   696→      const p = Math.min(1, (now - this.camTween.start) / this.camTween.duration);
   697→      const e = easeInOutCubic(p);
   698→      this.camera.position.lerpVectors(this.camTween.fromPos, this.camTween.toPos, e);
   699→      this.controls.target.lerpVectors(this.camTween.fromTarget, this.camTween.toTarget, e);
   700→      this.camera.lookAt(this.controls.target);
   701→      if (p >= 1) {
   702→        this.camTween = null;
   703→        this.controls.update();
   704→      }
   705→    }
   706→
   707→    // desvio vertical animado (enquadramento entre HUD e dock)
   708→    const prevShift = this.frameShift;
   709→    this.frameShift += (this.frameShiftTarget - this.frameShift) * 0.08;
   710→    if (Math.abs(this.frameShift - this.frameShiftTarget) > 0.05 && Math.abs(prevShift - this.frameShift) > 0.0005) {
   711→      const w = this.container.clientWidth || 1;
   712→      const h = this.container.clientHeight || 1;
   713→      if (this.frameShift > 0.5) {
   714→        this.camera.setViewOffset(w, h, 0, this.frameShift, w, h);
   715→      } else {
   716→        this.camera.clearViewOffset();
   717→      }
   718→    }
   719→
   720→    if (!this.anim && this.queue.length > 0) this.startNext();
   721→
   722→    if (this.anim) {
   723→      const p = Math.min(1, (now - this.anim.start) / this.anim.duration);
   724→      const eased =
   725→        this.anim.move.source === "user" || this.anim.move.source === "undo" || this.anim.move.source === "redo"
   726→          ? easeOutCubic(p)
   727→          : easeOutQuad(p);
   728→      const m = this.anim.move;
   729→      (this.pivot.rotation as unknown as Record<string, number>)[m.axis] = m.dir * HALF_PI * eased;
   730→      if (p >= 1) this.commitCurrent();
   731→    } else if (this.queue.length === 0 && this.queueWasActive) {
   732→      this.queueWasActive = false;
   733→      this.cb.onQueueEmpty();
   734→    }
   735→    if (this.queue.length > 0 || this.anim) this.queueWasActive = true;
   736→
   737→    // flutuação suave no menu
   738→    const targetBob = this.idle ? Math.sin(t * 0.0011) * 0.14 : 0;
   739→    this.cubeGroup.position.y += (targetBob - this.cubeGroup.position.y) * 0.06;
   740→    if (this.idle) {
   741→      this.cubeGroup.rotation.x = Math.sin(t * 0.0007) * 0.05;
   742→      this.cubeGroup.rotation.z = Math.cos(t * 0.0009) * 0.04;
   743→    } else {
   744→      this.cubeGroup.rotation.x *= 0.9;
   745→      this.cubeGroup.rotation.z *= 0.9;
   746→    }
   747→
   748→    // pulso de vitória
   749→    if (this.pulseStart > 0) {
   750→      const p = (now - this.pulseStart) / 900;
   751→      if (p >= 1) {
   752→        this.cubeGroup.scale.setScalar(1);
   753→        this.pulseStart = -1;
   754→      } else {
   755→        this.cubeGroup.scale.setScalar(1 + 0.075 * Math.sin(Math.PI * p));
   756→      }
   757→    }
   758→
   759→    this.controls.autoRotate = (this.idle || this.pulseStart > 0) && !this.camTween;
   760→    this.controls.update();
   761→    this.renderer.render(this.scene, this.camera);
   762→  };

   834→function snapAxis(v: THREE.Vector3): Vec3 {
   835→  const ax = Math.abs(v.x);
   836→  const ay = Math.abs(v.y);
   837→  const az = Math.abs(v.z);
   838→  if (ax >= ay && ax >= az) return [Math.sign(v.x), 0, 0];
   839→  if (ay >= az) return [0, Math.sign(v.y), 0];
   840→  return [0, 0, Math.sign(v.z)];
   841→}
   842→
   843→function axisOf(v: Vec3): Axis {
   844→  if (v[0] !== 0) return "x";
   845→  if (v[1] !== 0) return "y";
   846→  return "z";
   847→}

   849→function roundRect(g: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
   850→  g.beginPath();
   851→  g.moveTo(x + r, y);
   852→  g.arcTo(x + w, y, x + w, y + h, r);
   853→  g.arcTo(x + w, y + h, x, y + h, r);
   854→  g.arcTo(x, y + h, x, y, r);
   855→  g.arcTo(x, y, x + w, y, r);
   856→  g.closePath();
   857→}

   859→function lighten(hex: string, amt: number): string {
   860→  const c = new THREE.Color(hex);
   861→  if (amt >= 0) {
   862→    c.lerp(new THREE.Color(0xffffff), amt);
   863→  } else {
   864→    c.lerp(new THREE.Color(0x000000), -amt);
   865→  }
   866→  return `#${c.getHexString()}`;
   867→}
