   200→   200→  private makeStickerTexture(faceIndex: number): THREE.Texture {
   201→   201→    const key = `${this.theme.id}-${faceIndex}`;
   202→   202→    const cached = this.textureCache.get(key);
   203→   203→    if (cached) return cached;
   204→   204→    const canvas = document.createElement("canvas");
   205→   205→    canvas.width = 256;
   206→   206→    canvas.height = 256;
   207→   207→    const g = canvas.getContext("2d");
   208→   208→    const color = this.theme.faces[faceIndex];
   209→   209→    if (g) {
   210→   210→      const r = 62;
   211→   211→      const m = 10;
   212→   212→      g.clearRect(0, 0, 256, 256);
   213→   213→      // gradiente radial suave (brilho no topo-esquerda)
   214→   214→      const grad = g.createRadialGradient(96, 78, 20, 128, 128, 190);
   215→   215→      grad.addColorStop(0, lighten(color, 0.22));
   216→   216→      grad.addColorStop(0.55, color);
   217→   217→      grad.addColorStop(1, lighten(color, -0.14));
   218→   218→      g.fillStyle = grad;
   219→   219→      roundRect(g, m, m, 256 - 2 * m, 256 - 2 * m, r);
   220→   220→      g.fill();
   221→   221→      // contorno interior discreto
   222→   222→      g.strokeStyle = "rgba(0,0,0,0.18)";
   223→   223→      g.lineWidth = 5;
   224→   224→      roundRect(g, m + 3, m + 3, 256 - 2 * m - 6, 256 - 2 * m - 6, r - 4);
   225→   225→      g.stroke();
   226→   226→      // reflexo superior
   227→   227→      g.fillStyle = "rgba(255,255,255,0.13)";
   228→   228→      roundRect(g, m + 14, m + 12, 256 - 2 * m - 28, 70, 44);
   229→   229→      g.fill();
   230→   230→    }
   231→   231→    const tex = new THREE.CanvasTexture(canvas);
   232→   232→    tex.colorSpace = THREE.SRGBColorSpace;
   233→   233→    tex.anisotropy = 4;
   234→   234→    this.textureCache.set(key, tex);
   235→   235→    return tex;
   236→   236→  }

   259→   259→  private buildCube(): void {
   260→   260→    this.disposeCubeObjects();
   261→   261→    const n = this.size;
   262→   262→    const c = (n - 1) / 2;
   263→   263→    const plasticMat = new THREE.MeshStandardMaterial({
   264→   264→      color: new THREE.Color(this.theme.plastic),
   265→   265→      roughness: 0.34,
   266→   266→      metalness: 0.22,
   267→   267→    });
   268→   268→    for (let x = 0; x < n; x++) {
   269→   269→      for (let y = 0; y < n; y++) {
   270→   270→        for (let z = 0; z < n; z++) {
   271→   271→          const onSurface = x === 0 || x === n - 1 || y === 0 || y === n - 1 || z === 0 || z === n - 1;
   272→   272→          if (!onSurface) continue;
   273→   273→          const group = new THREE.Group();
   274→   274→          const body = new THREE.Mesh(this.bodyGeometry, plasticMat);
   275→   275→          group.add(body);
   276→   276→          const hitbox = new THREE.Mesh(this.hitboxGeometry, this.hitboxMaterial);
   277→   277→          group.add(hitbox);
   278→   278→          const pos: Vec3 = [x, y, z];
   279→   279→          const stickerDirs: Vec3[] = [];
   280→   280→          const dirs: Vec3[] = [
   281→   281→            [1, 0, 0],
   282→   282→            [-1, 0, 0],
   283→   283→            [0, 1, 0],
   284→   284→            [0, -1, 0],
   285→   285→            [0, 0, 1],
   286→   286→            [0, 0, -1],
   287→   287→          ];
   288→   288→          dirs.forEach((d, faceIndex) => {
   289→   289→            const onBoundary =
   290→   290→              (d[0] === 1 && x === n - 1) ||
   291→   291→              (d[0] === -1 && x === 0) ||
   292→   292→              (d[1] === 1 && y === n - 1) ||
   293→   293→              (d[1] === -1 && y === 0) ||
   294→   294→              (d[2] === 1 && z === n - 1) ||
   295→   295→              (d[2] === -1 && z === 0);
   296→   296→            if (!onBoundary) return;
   297→   297→            stickerDirs.push(d);
   298→   298→            const faceIndex2 =
   299→   299→              d[0] === 1 ? 0 : d[0] === -1 ? 1 : d[1] === 1 ? 2 : d[1] === -1 ? 3 : d[2] === 1 ? 4 : 5;
   300→   300→            const mat = new THREE.MeshStandardMaterial({
   301→   301→              map: this.makeStickerTexture(faceIndex2),
   302→   302→              roughness: 0.38,
   303→   303→              metalness: 0.05,
   304→   304→            });
   305→   305→            if (this.theme.glow > 0) {
   306→   306→              mat.emissive = new THREE.Color(0xffffff);
   307→   307→              mat.emissiveMap = mat.map ?? null;
   308→   308→              mat.emissiveIntensity = this.theme.glow;
   309→   309→            }
   310→   310→            const sticker = new THREE.Mesh(this.stickerGeometry, mat);
   311→   311→            const v = new THREE.Vector3(d[0], d[1], d[2]);
   312→   312→            sticker.position.set(d[0] * 0.476, d[1] * 0.476, d[2] * 0.476);
   313→   313→            sticker.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), v);
   314→   314→            sticker.userData.cubie = null; // preenchido abaixo
   315→   315→            group.add(sticker);
   316→   316→            this.stickerMeshes.push(sticker);
   317→   317→          });
   318→   318→          const cubie: Cubie = {
   319→   319→            pos,
   320→   320→            stickerDirs,
   321→   321→            M: [
   322→   322→              [1, 0, 0],
   323→   323→              [0, 1, 0],
   324→   324→              [0, 0, 1],
   325→   325→            ],
   326→   326→            group,
   327→   327→          };
   328→   328→          group.position.set((x - c) * this.spacing, (y - c) * this.spacing, (z - c) * this.spacing);
   329→   329→          group.userData.cubie = cubie;
   330→   330→          for (const child of group.children) {
   331→   331→            if (child instanceof THREE.Mesh && child !== body) child.userData.cubie = cubie;
   332→   332→          }
   333→   333→          this.hitMeshes.push(hitbox);
   334→   334→          this.cubies.push(cubie);
   335→   335→          this.cubeGroup.add(group);
   336→   336→        }
   337→   337→      }
   338→   338→    }
   339→   339→    this.buildShadow();
   340→   340→  }

   407→   407→  private applyThemeToMaterials(): void {
   408→   408→    for (const c of this.cubies) {
   409→   409→      c.group.traverse((o) => {
   410→   410→        const mesh = o as THREE.Mesh;
   411→   411→        if (!mesh.isMesh) return;
   412→   412→        const mat = mesh.material as THREE.MeshStandardMaterial;
   413→   413→        if (!mat || Array.isArray(mat)) return;
   414→   414→        if (mat.map) {
   415→   415→          if (this.blind) {
   416→   416→            // modo memória: adesivos neutros, sem brilho
   417→   417→            mat.map = this.makeBlindTexture();
   418→   418→            mat.emissive = new THREE.Color(0x000000);
   419→   419→            mat.emissiveMap = null;
   420→   420→            mat.emissiveIntensity = 0;
   421→   421→            mat.needsUpdate = true;
   422→   422→            return;
   423→   423→          }
   424→   424→          // adesivo: descobrir face pela orientação local
   425→   425→          const dir = new THREE.Vector3(0, 0, 1).applyQuaternion(mesh.quaternion);
   426→   426→          const faceIndex =
   427→   427→            dir.x > 0.5 ? 0 : dir.x < -0.5 ? 1 : dir.y > 0.5 ? 2 : dir.y < -0.5 ? 3 : dir.z > 0.5 ? 4 : 5;
   428→   428→          mat.map = this.makeStickerTexture(faceIndex);
   429→   429→          if (this.theme.glow > 0) {
   430→   430→            mat.emissive = new THREE.Color(0xffffff);
   431→   431→            mat.emissiveMap = mat.map;
   432→   432→            mat.emissiveIntensity = this.theme.glow;
   433→   433→          } else {
   434→   434→            mat.emissive = new THREE.Color(0x000000);
   435→   435→            mat.emissiveMap = null;
   436→   436→            mat.emissiveIntensity = 0;
   437→   437→          }
   438→   438→          mat.needsUpdate = true;
   439→   439→        } else {
   440→   440→          mat.color = new THREE.Color(this.theme.plastic);
   441→   441→        }
   442→   442→      });
   443→   443→    }
   444→   444→  }

   448→   448→  private fitCamera(): void {
   449→   449→    const r = this.idealCameraDistance();
   450→   450→    this.camera.position.set(r * 0.66, r * 0.5, r * 0.82);
   451→   451→    this.camera.lookAt(0, 0, 0);
   452→   452→    this.controls.target.set(0, 0, 0);
   453→   453→    this.controls.minDistance = this.size * 1.7 + 0.8;
   454→   454→    this.controls.maxDistance = r * 1.9;
   455→   455→    this.controls.update();
   456→   456→    this.updateFrameShiftTarget();

   459→   459→  /** Distância vertical (px) entre o centro do ecrã e o centro da zona livre entre HUD e dock. */
   460→   460→  private updateFrameShiftTarget(): void {
   461→   461→    if (!this.framed) {
   462→   462→      this.frameShiftTarget = 0;
   463→   463→      return;
   464→   464→    }
   465→   465→    const h = Math.max(1, this.container.clientHeight);
   466→   466→    // HUD superior ~8% da altura, dock inferior ~13% (com mínimos em px)
   467→   467→    const top = Math.max(56, h * 0.085);
   468→   468→    const bottom = Math.max(96, h * 0.14);
   469→   469→    this.frameShiftTarget = Math.max(0, (bottom - top) / 2);
   470→   470→  }

   472→   472→  /**
   473→   473→   * Repõe a câmara: centro (0,0,0), zoom padrão e orientação de origem, com
   474→   474→   * animação suave. Chamado pelo botão "Centrar" e por duplo clique no fundo.
   475→   475→   */
   476→   476→  resetView(): void {
   477→   477→    const r = this.idealCameraDistance();
   478→   478→    this.defaultCamPos.set(r * 0.66, r * 0.5, r * 0.82);
   479→   479→    this.controls.minDistance = this.size * 1.7 + 0.8;
   480→   480→    this.controls.maxDistance = r * 1.9;
   481→   481→    this.camTween = {
   482→   482→      fromPos: this.camera.position.clone(),
   483→   483→      toPos: this.defaultCamPos.clone(),
   484→   484→      fromTarget: this.controls.target.clone(),
   485→   485→      toTarget: new THREE.Vector3(0, 0, 0), // o offset visual (setViewOffset) trata do enquadramento
   486→   486→      start: performance.now(),
   487→   487→      duration: 650,
   488→   488→    };
   489→   489→    // repe a deriva do grupo do cubo (flutuação/rotação do menu)
   490→   490→    this.cubeGroup.position.y = 0;
   491→   491→    this.cubeGroup.rotation.x = 0;
   492→   492→    this.cubeGroup.rotation.z = 0;
   493→   493→  }

   495→   495→  /** Distância ideal da câmara para o tamanho do cubo + aspeto atual do contentor. */
   496→   496→  private idealCameraDistance(): number {
   497→   497→    const s = this.size;
   498→   498→    const base = s * 2.45 + 1.75;
   499→   499→    const aspect = this.container.clientWidth / Math.max(1, this.container.clientHeight);
   500→   500→    const halfW = (s * 1.04) / 2 + 0.25;
   501→   501→    const hTan = Math.tan((this.camera.fov * Math.PI) / 360) * Math.max(0.3, aspect);
   502→   502→    const need = (halfW / hTan) * 1.06;
   503→   503→    return Math.min(Math.max(base, need), s * 7 + 4);
   504→   504→  }

   532→   532→  private updatePointerNDC(e: PointerEvent): void {
   533→   533→    const rect = this.renderer.domElement.getBoundingClientRect();
   534→   534→    this.pointer.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
   535→   535→    this.pointer.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;

   538→   538→  private onPointerDown = (e: PointerEvent): void => {
   539→   539→    if (e.button !== 0 && e.pointerType === "mouse") return;
   540→   540→    if (this.anim || this.queue.length > 0) return;
   541→   541→    this.updatePointerNDC(e);
   542→   542→    this.raycaster.setFromCamera(this.pointer, this.camera);
   543→   543→    const hits = this.raycaster.intersectObjects(this.hitMeshes, false);
   544→   544→    if (hits.length === 0) return;
   545→   545→    const hit = hits[0];
   546→   546→    const cubie = hit.object.userData.cubie as Cubie | undefined;
   547→   547→    if (!cubie) return;
   548→   548→    // normal da face da hitbox -> espaço do mundo
   549→   549→    const worldNormal = hit.face
   550→   550→      ? hit.face.normal.clone().transformDirection(hit.object.matrixWorld)
   551→   551→      : hit.object.getWorldDirection(new THREE.Vector3());
   552→   552→    const snapped = snapAxis(worldNormal);
   553→   553→    this.drag = {
   554→   554→      x0: e.clientX,
   555→   555→      y0: e.clientY,
   556→   556→      cubie,
   557→   557→      normal: snapped,
   558→   558→      point: hit.point.clone(),
   559→   559→      decided: false,
   560→   560→    };
   561→   561→    this.controls.enabled = false;
   562→   562→  };
   563→   563→
   564→   564→  private onPointerMove = (e: PointerEvent): void => {
   565→   565→    if (!this.drag || this.drag.decided) return;
   566→   566→    const dx = e.clientX - this.drag.x0;
   567→   567→    const dy = e.clientY - this.drag.y0;
   568→   568→    if (Math.hypot(dx, dy) < 12) return;
   569→   569→
   570→   570→    // vetor de arrasto no mundo (plano do adesivo)
   571→   571→    this.updatePointerNDC(e);
   572→   572→    this.raycaster.setFromCamera(this.pointer, this.camera);
   573→   573→    const n = new THREE.Vector3(...this.drag.normal);
   574→   574→    const plane = new THREE.Plane().setFromNormalAndCoplanarPoint(n, this.drag.point);
   575→   575→    const pNow = new THREE.Vector3();
   576→   576→    let delta: THREE.Vector3 | null = this.raycaster.ray.intersectPlane(plane, pNow)
   577→   577→      ? pNow.clone().sub(this.drag.point)
   578→   578→      : null;
   579→   579→    if (!delta || delta.lengthSq() < 1e-8) {
   580→   580→      // plano quase de perfil: aproximar pelo ecrã
   581→   581→      const dist = this.camera.position.distanceTo(this.drag.point);
   582→   582→      const k = dist * 0.0016;
   583→   583→      const right = new THREE.Vector3().setFromMatrixColumn(this.camera.matrixWorld, 0);
   584→   584→      const up = new THREE.Vector3().setFromMatrixColumn(this.camera.matrixWorld, 1);
   585→   585→      delta = right.multiplyScalar(dx * k).add(up.multiplyScalar(-dy * k));
   586→   586→    }
   587→   587→    delta.normalize();
   588→   588→
   589→   589→    // eixos candidatos: os dois perpendiculares à normal do adesivo
   590→   590→    const normalAxis = axisOf(this.drag.normal);
   591→   591→    const r = this.drag.point.clone().sub(this.cubeGroup.position);
   592→   592→    let best: { axis: Axis; dir: 1 | -1; score: number } | null = null;
   593→   593→    for (const axis of ["x", "y", "z"] as Axis[]) {
   594→   594→      if (axis === normalAxis) continue;
   595→   595→      for (const dir of [1, -1] as const) {
   596→   596→        const omega = AXIS_VEC[axis].clone().multiplyScalar(dir);
   597→   597→        const v = new THREE.Vector3().crossVectors(omega, r).normalize();
   598→   598→        const score = v.dot(delta);
   599→   599→        if (!best || score > best.score) best = { axis, dir, score };
   600→   600→      }
   601→   601→    }
   602→   602→    this.drag.decided = true;
   603→   603→    if (best && best.score > 0.12) {
   604→   604→      const layer = this.drag.cubie.pos[AXIS_INDEX[best.axis]];
   605→   605→      this.cb.onUserGestureMove({
   606→   606→        axis: best.axis,
   607→   607→        layer,
   608→   608→        dir: best.dir,
   609→   609→        source: "user",
   610→   610→      });
   611→   611→    }
   612→   612→  };

   642→   642→
   643→   643→  private startNext(): void {
   644→   644→    const move = this.queue.shift();
   645→   645→    if (!move) return;
   646→   646→    const ai = AXIS_INDEX[move.axis];
   647→   647→    const members = this.cubies.filter((c) => c.pos[ai] === move.layer);
   648→   648→    this.pivot.rotation.set(0, 0, 0);
   649→   649→    this.pivot.updateMatrixWorld(true);
   650→   650→    for (const c of members) this.pivot.attach(c.group);
   651→   651→    this.anim = {
   652→   652→      move,
   653→   653→      cubies: members,
   654→   654→      start: performance.now(),
   655→   655→      duration: move.duration ?? defaultDuration(move, this.queue.length),
   656→   656→    };
   657→   657→    this.cb.onMoveStart(move);
   658→   658→  }
   659→   659→
   660→   660→  private commitCurrent(): void {
   661→   661→    const { move, cubies } = this.anim!;
   662→   662→    this.pivot.rotation.set(0, 0, 0);
   663→   663→    (this.pivot.rotation as unknown as Record<string, number>)[move.axis] = move.dir * HALF_PI;
   664→   664→    this.pivot.updateMatrixWorld(true);
   665→   665→    const R = rotationMatrix(move.axis, move.dir);
   666→   666→    const c = (this.size - 1) / 2;
   667→   667→    const m4 = new THREE.Matrix4();
   668→   668→    for (const cubie of cubies) {
   669→   669→      this.cubeGroup.attach(cubie.group);
   670→   670→      cubie.pos = rotateLogicalPos(cubie.pos, move.axis, move.dir, this.size);
   671→   671→      cubie.M = matMul(R, cubie.M);
   672→   672→      // encaixe exato (elimina deriva numérica)
   673→   673→      cubie.group.position.set(
   674→   674→        (cubie.pos[0] - c) * this.spacing,
   675→   675→        (cubie.pos[1] - c) * this.spacing,
   676→   676→        (cubie.pos[2] - c) * this.spacing
   677→   677→      );
   678→   678→      const M = cubie.M;
   679→   679→      m4.set(M[0][0], M[0][1], M[0][2], 0, M[1][0], M[1][1], M[1][2], 0, M[2][0], M[2][1], M[2][2], 0, 0, 0, 0, 1);
   680→   680→      cubie.group.quaternion.setFromRotationMatrix(m4);
   681→   681→      cubie.group.scale.set(1, 1, 1);
   682→   682→    }
   683→   683→    this.pivot.rotation.set(0, 0, 0);
   684→   684→    const committed = this.anim!.move;
   685→   685→    this.anim = null;
   686→   686→    this.cb.onMoveCommitted(committed);
   687→   687→  }

   689→   689→  private tick = (): void => {
   690→   690→    if (this.disposed) return;
   691→   691→    const now = performance.now();
   692→   692→    const t = now - this.clockStart;
   693→   693→
   694→   694→    // tween da câmara (botão centrar / duplo clique)
   695→   695→    if (this.camTween) {
   696→   696→      const p = Math.min(1, (now - this.camTween.start) / this.camTween.duration);
   697→   697→      const e = easeInOutCubic(p);
   698→   698→      this.camera.position.lerpVectors(this.camTween.fromPos, this.camTween.toPos, e);
   699→   699→      this.controls.target.lerpVectors(this.camTween.fromTarget, this.camTween.toTarget, e);
   700→   700→      this.camera.lookAt(this.controls.target);
   701→   701→      if (p >= 1) {
   702→   702→        this.camTween = null;
   703→   703→        this.controls.update();
   704→   704→      }
   705→   705→    }
   706→   706→
   707→   707→    // desvio vertical animado (enquadramento entre HUD e dock)
   708→   708→    const prevShift = this.frameShift;
   709→   709→    this.frameShift += (this.frameShiftTarget - this.frameShift) * 0.08;
   710→   710→    if (Math.abs(this.frameShift - this.frameShiftTarget) > 0.05 && Math.abs(prevShift - this.frameShift) > 0.0005) {
   711→   711→      const w = this.container.clientWidth || 1;
   712→   712→      const h = this.container.clientHeight || 1;
   713→   713→      if (this.frameShift > 0.5) {
   714→   714→        this.camera.setViewOffset(w, h, 0, this.frameShift, w, h);
   715→   715→      } else {
   716→   716→        this.camera.clearViewOffset();
   717→   717→      }
   718→   718→    }
   719→   719→
   720→   720→    if (!this.anim && this.queue.length > 0) this.startNext();
   721→   721→
   722→   722→    if (this.anim) {
   723→   723→      const p = Math.min(1, (now - this.anim.start) / this.anim.duration);
   724→   724→      const eased =
   725→   725→        this.anim.move.source === "user" || this.anim.move.source === "undo" || this.anim.move.source === "redo"
   726→   726→          ? easeOutCubic(p)
   727→   727→          : easeOutQuad(p);
   728→   728→      const m = this.anim.move;
   729→   729→      (this.pivot.rotation as unknown as Record<string, number>)[m.axis] = m.dir * HALF_PI * eased;
   730→   730→      if (p >= 1) this.commitCurrent();
   731→   731→    } else if (this.queue.length === 0 && this.queueWasActive) {
   732→   732→      this.queueWasActive = false;
   733→   733→      this.cb.onQueueEmpty();
   734→   734→    }
   735→   735→    if (this.queue.length > 0 || this.anim) this.queueWasActive = true;
   736→   736→
   737→   737→    // flutuação suave no menu
   738→   738→    const targetBob = this.idle ? Math.sin(t * 0.0011) * 0.14 : 0;
   739→   739→    this.cubeGroup.position.y += (targetBob - this.cubeGroup.position.y) * 0.06;
   740→   740→    if (this.idle) {
   741→   741→      this.cubeGroup.rotation.x = Math.sin(t * 0.0007) * 0.05;
   742→   742→      this.cubeGroup.rotation.z = Math.cos(t * 0.0009) * 0.04;
   743→   743→    } else {
   744→   744→      this.cubeGroup.rotation.x *= 0.9;
   745→   745→      this.cubeGroup.rotation.z *= 0.9;
   746→   746→    }
   747→   747→
   748→   748→    // pulso de vitória
   749→   749→    if (this.pulseStart > 0) {
   750→   750→      const p = (now - this.pulseStart) / 900;
   751→   751→      if (p >= 1) {
   752→   752→        this.cubeGroup.scale.setScalar(1);
   753→   753→        this.pulseStart = -1;
   754→   754→      } else {
   755→   755→        this.cubeGroup.scale.setScalar(1 + 0.075 * Math.sin(Math.PI * p));
   756→   756→      }
   757→   757→    }
   758→   758→
   759→   759→    this.controls.autoRotate = (this.idle || this.pulseStart > 0) && !this.camTween;
   760→   760→    this.controls.update();
   761→   761→    this.renderer.render(this.scene, this.camera);
   762→   762→  };

   834→   834→function snapAxis(v: THREE.Vector3): Vec3 {
   835→   835→  const ax = Math.abs(v.x);
   836→   836→  const ay = Math.abs(v.y);
   837→   837→  const az = Math.abs(v.z);
   838→   838→  if (ax >= ay && ax >= az) return [Math.sign(v.x), 0, 0];
   839→   839→  if (ay >= az) return [0, Math.sign(v.y), 0];
   840→   840→  return [0, 0, Math.sign(v.z)];
   841→   841→}

   843→   843→function axisOf(v: Vec3): Axis {
   844→   844→  if (v[0] !== 0) return "x";
   845→   845→  if (v[1] !== 0) return "y";
   846→   846→  return "z";
   847→   847→}

   849→   849→function roundRect(g: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
   850→   850→  g.beginPath();
   851→   851→  g.moveTo(x + r, y);
   852→   852→  g.arcTo(x + w, y, x + w, y + h, r);
   853→   853→  g.arcTo(x + w, y + h, x, y + h, r);
   854→   854→  g.arcTo(x, y + h, x, y, r);
   855→   855→  g.arcTo(x, y, x + w, y, r);
